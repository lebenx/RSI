"""Replayable ScienceWorld planner comparison.

This is a diagnostic interactive benchmark, not a claim of reproducing an
end-to-end LLM agent. Candidate rollouts are generated by real simulator resets
and actions. A shared premise is the task/variation/decision snapshot. Flat and
provenance planners receive exactly the same bank; only duplicate normalization
differs. Gold paths are used offline for labels and regret, never as planner
input.
"""
from __future__ import annotations
import argparse, csv, hashlib, json, random, time
from pathlib import Path
import numpy as np

TASKS = ["boil", "change-the-state-of-matter-of", "power-component", "find-living-thing", "test-conductivity"]
METHODS = ("flat", "provenance")
CONDITIONS = ("pure_duplication", "independent_rollout")
MULTIPLICITIES = (1, 2, 4, 8)

def stable_int(s): return int(hashlib.sha256(s.encode()).hexdigest()[:12], 16)

class Replay:
    def __init__(self, env, task, variation, simplification="easy", limit=35):
        self.env, self.task, self.variation = env, task, variation
        self.simplification, self.limit = simplification, limit
    def reset(self):
        self.env.load(self.task, self.variation, self.simplification, generateGoldPath=True)
        obs, info = self.env.reset()
        return obs, info
    def replay(self, prefix):
        obs, info = self.reset()
        for action in prefix:
            obs, reward, done, info = self.env.step(action)
            if done: break
        return obs, info
    def action_value(self, prefix, action, followups=()):
        self.replay(prefix)
        obs, reward, done, info = self.env.step(action)
        total = float(reward); score = float(info["score"]); actions = [action]
        for next_action in followups:
            if done: break
            obs, reward, done, info = self.env.step(next_action)
            total += float(reward); score = float(info["score"]); actions.append(next_action)
        return {"value": total, "score": score, "done": bool(done), "actions": actions,
                "observation": obs[:500], "valid_after": info.get("valid", [])[:20]}

def action_candidates(info, gold_action, max_candidates=6):
    valid = [a for a in info.get("valid", []) if isinstance(a, str) and a and "look around" != a]
    out=[]
    if gold_action in valid: out.append(gold_action)
    for a in valid:
        if a not in out: out.append(a)
        if len(out) >= max_candidates: break
    return out

def build_bank(replay, prefix, info, gold_action, condition, m, decision_id, max_candidates=6):
    candidates=action_candidates(info,gold_action,max_candidates)
    base=[]
    premise=f"{replay.task}:{replay.variation}:decision:{decision_id}"
    for idx, action in enumerate(candidates):
        if condition == "pure_duplication":
            ev=replay.action_value(prefix, action)
            base.append({"sample_id":f"{premise}:a{idx}:base","source_id":premise,
                         "premise_id":premise,"action":action,"value":ev["value"],
                         "score":ev["score"],"followups":[],"text":f"{action} yields {ev['value']:+.3f}."})
        else:
            first=replay.action_value(prefix, action)
            follow=first["valid_after"] or ["look around"]
            for j in range(m):
                f=(follow[j % len(follow)],)
                ev=replay.action_value(prefix, action, f)
                base.append({"sample_id":f"{premise}:a{idx}:draw{j}","source_id":premise,
                             "premise_id":premise,"action":action,"value":ev["value"],
                             "score":ev["score"],"followups":list(f),
                             "text":f"{action} then {f[0]} yields {ev['value']:+.3f}."})
    if condition == "pure_duplication" and m > 1 and base:
        gold_rows=[r for r in base if r["action"]==gold_action]
        non_gold=[r for r in base if r["action"]!=gold_action]
        target=max(non_gold or base,key=lambda r:r["value"])
        for j in range(m-1):
            copy=dict(target); copy["sample_id"]=target["sample_id"]
            copy["copy_index"]=j+1; base.append(copy)
    return base, candidates

def choose(bank, method):
    groups={}
    for r in bank: groups.setdefault(r["action"],[]).append(r)
    scores={}
    for action, rows in groups.items():
        if method=="flat": scores[action]=sum(r["value"] for r in rows)
        else:
            by_id={r["sample_id"]:r for r in rows}
            scores[action]=float(np.mean([r["value"] for r in by_id.values()]))
    action=max(scores,key=scores.get)
    return action,scores

def oracle_run(replay, gold, limit):
    replay.reset(); total=0.; trace=[]; done=False; score=0
    for action in gold[:limit]:
        obs,reward,done,info=replay.env.step(action); total += float(reward); score=float(info["score"]); trace.append(action)
        if done: break
    return {"score":score,"reward":total,"done":bool(done),"actions":trace}

def run_episode(ScienceWorldEnv, task, variation, method, condition, m, horizon):
    env=ScienceWorldEnv(envStepLimit=max(35,horizon+5)); replay=Replay(env,task,variation,limit=horizon+5)
    obs, info=replay.reset(); gold=env.get_gold_action_sequence(); prefix=[]; total_reward=0.; step_trace=[]
    for t in range(horizon):
        gold_action=gold[t] if t < len(gold) else None
        bank,candidates=build_bank(replay,prefix,info,gold_action,condition,m,f"t{t}")
        chosen,scores=choose(bank,method)
        obs,reward,done,info=env.step(chosen); total_reward += float(reward); prefix.append(chosen)
        step_trace.append({"t":t,"gold_action":gold_action,"chosen_action":chosen,
                           "candidates":candidates,"scores":scores,"bank":bank,
                           "reward":float(reward),"score":float(info.get("score",0)),
                           "source_groups":len({r["source_id"] for r in bank}),
                           "unique_samples":len({r["sample_id"] for r in bank}),
                           "presented_samples":len(bank)})
        if done: break
    oracle=oracle_run(replay,gold,horizon)
    success=int(info.get("score",0)>0)
    unnecessary=sum(int(x["chosen_action"]!=x["gold_action"]) for x in step_trace if x["gold_action"] is not None)
    env.close()
    return {"task":task,"variation":variation,"method":method,"condition":condition,"multiplicity":m,
            "success":success,"reward":total_reward,"score":float(info.get("score",0)),
            "oracle_score":oracle["score"],"regret":oracle["score"]-float(info.get("score",0)),
            "unnecessary_actions":unnecessary,"steps":len(step_trace),"rollout_count":sum(x["presented_samples"] for x in step_trace),
            "unique_samples":sum(x["unique_samples"] for x in step_trace),"trace":step_trace}

def run(output, tasks=TASKS, variations=2, horizon=3):
    from scienceworld import ScienceWorldEnv
    out=Path(output);out.mkdir(parents=True,exist_ok=True); rows=[]; traces=[]; started=time.time()
    for task in tasks:
        probe=ScienceWorldEnv(envStepLimit=40)
        probe.load(task,0,"easy",generateGoldPath=True); available=probe.get_variations_test()[:variations]; probe.close()
        for variation in available:
            for condition in CONDITIONS:
                for m in MULTIPLICITIES:
                    for method in METHODS:
                        result=run_episode(ScienceWorldEnv,task,variation,method,condition,m,horizon)
                        trace=result.pop("trace"); rows.append(result); traces.append({**result,"trace":trace})
                        print(task,variation,condition,m,method,result["score"],flush=True)
    frame_rows=[{k:v for k,v in r.items()} for r in rows]
    fields=list(frame_rows[0])
    with (out/"scienceworld_raw.csv").open("w") as f:
        writer=csv.DictWriter(f,fieldnames=fields);writer.writeheader();writer.writerows(frame_rows)
    (out/"scienceworld_traces.jsonl").write_text("\n".join(json.dumps(x,ensure_ascii=False) for x in traces)+"\n")
    import pandas as pd
    df=pd.DataFrame(frame_rows); summary=[]
    for keys,g in df.groupby(["condition","multiplicity","method"]):
        summary.append({"condition":keys[0],"multiplicity":keys[1],"method":keys[2],"n":len(g),
                        "success_rate":g.success.mean(),"reward_mean":g.reward.mean(),"score_mean":g.score.mean(),
                        "regret_mean":g.regret.mean(),"unnecessary_actions_mean":g.unnecessary_actions.mean(),
                        "rollout_count_mean":g.rollout_count.mean(),"unique_samples_mean":g.unique_samples.mean()})
    pd.DataFrame(summary).to_csv(out/"scienceworld_summary.csv",index=False)
    meta={"tasks":list(tasks),"variations_per_task":variations,"horizon":horizon,"conditions":list(CONDITIONS),
          "multiplicities":list(MULTIPLICITIES),"methods":list(METHODS),
          "scienceworld_commit":"e8216d6044e8e39be9fcb185e3b2dfb602584b52","seconds":time.time()-started}
    (out/"scienceworld_metadata.json").write_text(json.dumps(meta,indent=2))
    print(json.dumps(meta,indent=2)); return meta

if __name__=="__main__":
    p=argparse.ArgumentParser();p.add_argument("--output",default="results_submission/scienceworld");p.add_argument("--variations",type=int,default=2);p.add_argument("--horizon",type=int,default=3);a=p.parse_args();run(a.output,variations=a.variations,horizon=a.horizon)
